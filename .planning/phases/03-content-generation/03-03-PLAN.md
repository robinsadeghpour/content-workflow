**Step 6 — Generate TikTok EN slide content via Claude:**
Call Claude API to generate slide texts for TikTok. The prompt must:
- Include the idea title, summary, and transcript (if available)
- Include the voice-casual.xml profile
- Request a JSON response with: `{ "slides": [{ "text": "...", "photo_description": "..." }], "caption": "...", "hook_formula": "..." }`
- Slide count is flexible per D-04 (instruct Claude: "Use 3-5 slides for hot takes, 5-8 for tutorials/listicles, adapt to content")
- Each slide text must follow rules from add-text-overlay.js: no emoji, use \n for line breaks, keep lines to 4-6 words
- Each `photo_description` is used for photo catalog matching or AI prompt generation

```javascript
const tiktokResponse = await anthropic.messages.create({
  model: 'claude-sonnet-4-20250514',
  max_tokens: 2000,
  system: `You are Robin Faraj creating TikTok slideshow content. Voice profile:\n${voiceCasual}\n\nRules:\n- No emoji in slide text (canvas cannot render them)\n- Use \\n for manual line breaks\n- Keep lines to 4-6 words for readability\n- Slide count: 3-5 for hot takes, 5-8 for tutorials/listicles\n- Return valid JSON only`,
  messages: [{ role: 'user', content: `Create TikTok slideshow slides for this topic:\n\nTitle: ${idea.title}\nSummary: ${idea.summary}\nTranscript: ${idea.transcript || 'None'}\nSource: ${idea.source_url || 'None'}\n\nReturn JSON: { "slides": [{ "text": "slide text with\\nline breaks", "photo_description": "description for photo matching" }], "caption": "TikTok caption text", "hook_formula": "name of hook formula used" }` }]
});
const tiktokContent = JSON.parse(tiktokResponse.content[0].text);
```

**Step 7 — Match photos from catalog or generate AI slides:**
```javascript
if (visualApproach === 'photo_overlay' && catalog.photos.length > 0) {
  // Match photos by keyword overlap between slide photo_description and catalog photo descriptions
  const selectedPhotos = tiktokContent.slides.map(slide => {
    const keywords = slide.photo_description.toLowerCase().split(/\s+/);
    let bestMatch = catalog.photos[0]; // fallback to first photo
    let bestScore = 0;
    for (const photo of catalog.photos) {
      const desc = (photo.description + ' ' + photo.mood + ' ' + photo.best_for.join(' ')).toLowerCase();
      const score = keywords.filter(kw => desc.includes(kw)).length;
      if (score > bestScore) { bestScore = score; bestMatch = photo; }
    }
    return path.join(__dirname, '..', 'media/images/tiktok', bestMatch.filename);
  });
  // Write photos.json and texts.json for generate-tiktok-slides.js
  const photosJson = path.join(outputBase, 'tiktok-en', 'photos.json');
  const textsJson = path.join(outputBase, 'tiktok-en', 'texts.json');
  fs.writeFileSync(photosJson, JSON.stringify(selectedPhotos));
  fs.writeFileSync(textsJson, JSON.stringify(tiktokContent.slides.map(s => s.text)));
  // Call generate-tiktok-slides.js
  spawnSync('node', [path.join(__dirname, 'generate-tiktok-slides.js'), '--photos', photosJson, '--texts', textsJson, '--output', path.join(outputBase, 'tiktok-en')], { stdio: 'inherit' });
} else {
  // AI generation fallback
  const promptsJson = path.join(outputBase, 'tiktok-en', 'prompts.json');
  fs.writeFileSync(promptsJson, JSON.stringify(tiktokContent.slides.map(s => `Portrait photo for TikTok slide, 1080x1440, ${s.photo_description}`)));
  spawnSync('node', [path.join(__dirname, 'generate-ai-slides.js'), '--prompts', promptsJson, '--output', path.join(outputBase, 'tiktok-en')], { stdio: 'inherit' });
  // Now overlay text on AI-generated images
  const aiFiles = tiktokContent.slides.map((_, i) => path.join(outputBase, 'tiktok-en', `slide-${String(i + 1).padStart(2, '0')}.png`));
  const photosJson = path.join(outputBase, 'tiktok-en', 'photos-ai.json');
  const textsJson = path.join(outputBase, 'tiktok-en', 'texts.json');
  fs.writeFileSync(photosJson, JSON.stringify(aiFiles));
  fs.writeFileSync(textsJson, JSON.stringify(tiktokContent.slides.map(s => s.text)));
  spawnSync('node', [path.join(__dirname, 'generate-tiktok-slides.js'), '--photos', photosJson, '--texts', textsJson, '--output', path.join(outputBase, 'tiktok-en')], { stdio: 'inherit' });
}
```
